{{- $id := .Get "id" | default (.Get 0) -}} {{- if not $id -}} {{- errorf "The %q shortcode requires a video ID. See %s" .Name .Position -}} {{- end -}} {{- $privacy := site.Config.Privacy.YouTube -}} {{- $start := .Get "start" -}} {{- $end := .Get "end" -}} {{- $caption := .Get "caption" -}} {{- $title := .Get "title" -}} {{- if not $title -}} {{- $oembed := printf "https://www.youtube.com/oembed?format=json&url=%s" (printf "https://www.youtube.com/watch?v=%s" $id | urlquery) -}} {{- with try (resources.GetRemote $oembed) -}} {{- with .Err -}} {{- warnf "youtube: unable to get the title of %q: %s" $id . -}} {{- else with .Value -}} {{- $title = (. | transform.Unmarshal).title -}} {{- end -}} {{- end -}} {{- end -}} {{- $title = $title | default "YouTube" -}} {{- $thumbnail := false -}} {{- range slice "maxresdefault" "hqdefault" -}} {{- if not $thumbnail -}} {{- with try (resources.GetRemote (printf "https://i.ytimg.com/vi/%s/%s.jpg" $id .)) -}} {{- with .Value -}} {{- $thumbnail = .Fill "1280x720 webp q80" | resources.Copy (printf "img/youtube/%s.webp" $id) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $watch := printf "https://www.youtube.com/watch?v=%s" $id -}} {{- with $start }}{{ $watch = printf "%s&t=%vs" $watch . }}{{ end -}} {{- $query := dict "autoplay" 1 "rel" 0 -}} {{- with $start }}{{ $query = merge $query (dict "start" .) }}{{ end -}} {{- with $end }}{{ $query = merge $query (dict "end" .) }}{{ end -}} {{- $host := cond $privacy.PrivacyEnhanced "www.youtube-nocookie.com" "www.youtube.com" -}} {{- $embed := printf "https://%s/embed/%s?%s" $host $id (collections.Querify $query) -}}
